cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD seeed_xiao_rp2350 CACHE STRING "Controller board")
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(box_r21r_rp C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(box_r21r_rp main.c ../common/power.c ../common/protocol.c)
target_include_directories(box_r21r_rp PRIVATE ../common)
target_compile_options(box_r21r_rp PRIVATE -Wall -Wextra -Werror)
target_link_libraries(box_r21r_rp pico_stdlib pico_rand pico_i2c_slave
                      hardware_i2c hardware_uart hardware_sync)
pico_enable_stdio_uart(box_r21r_rp 0)
pico_enable_stdio_usb(box_r21r_rp 0)
pico_add_extra_outputs(box_r21r_rp)
